/** * ARDIE - API Google Sheets Backend (Version Robuste & Auto-réparatrice) */ function doGet(e) { try { var action = (e && e.parameter) ? e.parameter.action : null; if (action === 'getConfig') { var config = fetchConfigData(); return ContentService.createTextOutput(JSON.stringify(config)) .setMimeType(ContentService.MimeType.JSON); } if (action === 'getReports') { var reports = fetchReportsData(e.parameter.type, e.parameter.serre, e.parameter.startDate, e.parameter.endDate); return ContentService.createTextOutput(JSON.stringify(reports)) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ status: "online", time: new Date() })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ error: true, message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(15000); try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); if (data.type === 'CLIMAT') { var sheet = getOrCreateSheet(ss, 'Releves_Climatiques', [ 'Timestamp', 'Date', 'Heure', 'Serre', 'Intervenant', 'Temperature', 'Humidite_Air', 'Lumens', 'Humidite_Terreau', 'PH_Terreau', 'Eclairage', 'Arrosage', 'Remarques' ]); sheet.appendRow([ new Date().toISOString(), data.date, data.heure, data.serre, data.intervenant, Number(data.temperature), Number(data.humidite_air), data.lumens === '-' ? '' : Number(data.lumens), data.humidite_terreau === '-' ? '' : Number(data.humidite_terreau), data.ph === '-' ? '' : Number(data.ph), data.eclairage, data.arrosage, data.remarque ]); } else if (data.type === 'SUIVI_PLANTS') { var sheet = getOrCreateSheet(ss, 'Suivi_Plants', [ 'Timestamp', 'Date', 'Heure', 'Serre', 'Intervenant', 'Nb_Initial', 'Nb_Isoles', 'Nb_Morts', 'Nb_Surveillance', 'Nb_Sains', 'Observations' ]); sheet.appendRow([ new Date().toISOString(), data.date, data.heure, data.serre, data.intervenant, Number(data.initial), Number(data.isoles), Number(data.morts), Number(data.surveillance), Number(data.sains), data.observations ]); } else if (data.type === 'MANAGE_LOCATION') { var sheet = getOrCreateSheet(ss, 'Config_Emplacements', ['Nom_Emplacement', 'Statut']); if (data.subAction === 'ADD') { sheet.appendRow([data.name, 'Actif']); } else if (data.subAction === 'DELETE') { var rows = sheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { if (rows[i][0] === data.name) { sheet.deleteRow(i + 1); break; } } } } else if (data.type === 'MANAGE_AGENT') { var sheet = getOrCreateSheet(ss, 'Config_Agents', ['ID_SharePoint', 'Civilite', 'Nom_Prenom', 'Statut_Contrat', 'Atelier', 'Statut_Activite']); if (data.subAction === 'ADD') { sheet.appendRow(['LOCAL_' + Date.now(), '', data.name, 'Technicien', 'Serre', 'Actif']); } else if (data.subAction === 'TOGGLE_STATUS') { var rows = sheet.getDataRange().getValues(); for (var j = 1; j < rows.length; j++) { if (rows[j][2] === data.name) { // Colonne C: Nom_Prenom var currentStatus = rows[j][5]; // Colonne F: Statut_Activite var nextStatus = (currentStatus === 'Actif') ? 'Inactif' : 'Actif'; sheet.getRange(j + 1, 6).setValue(nextStatus); break; } } } } return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } // Récupération sécurisée sans plantage function fetchConfigData() { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Emplacements (Créé automatiquement si absent) var empSheet = getOrCreateSheet(ss, 'Config_Emplacements', ['Nom_Emplacement', 'Statut']); var empRows = empSheet.getDataRange().getValues(); var emplacements = []; if (empRows.length <= 1) { // Initialisation minimale si la feuille est vide var defEmp = ['Serre principale', 'Serre 1', 'Serre 2']; defEmp.forEach(function(e) { empSheet.appendRow([e, 'Actif']); }); emplacements = defEmp; } else { for (var i = 1; i < empRows.length; i++) { if (empRows[i][1] === 'Actif' && empRows[i][0]) { emplacements.push(empRows[i][0]); } } } // 2. Agents SharePoint var agSheet = getOrCreateSheet(ss, 'Config_Agents', ['ID_SharePoint', 'Civilite', 'Nom_Prenom', 'Statut_Contrat', 'Atelier', 'Statut_Activite']); var agRows = agSheet.getDataRange().getValues(); var agentsActifs = []; var tousLesAgents = []; for (var j = 1; j < agRows.length; j++) { var idSp = agRows[j][0]; var civ = agRows[j][1] || ''; var nom = agRows[j][2]; // Nom_Prenom var contrat = agRows[j][3] || ''; var atelier = agRows[j][4] || ''; var statutAct = agRows[j][5] || 'Actif'; if (nom) { var displayName = (civ ? civ + ' ' : '') + nom; tousLesAgents.push({ id: idSp, name: nom, displayName: displayName, contrat: contrat, atelier: atelier, status: statutAct }); // Filtre uniquement les agents Actifs if (statutAct === 'Actif') { agentsActifs.push({ nom: nom, displayName: displayName, label: displayName + (contrat ? ' (' + contrat + ')' : '') }); } } } return { emplacements: emplacements, agentsActifs: agentsActifs, tousLesAgents: tousLesAgents }; } function fetchReportsData(type, serreFilter, startDate, endDate) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(type === 'PLANTS' ? 'Suivi_Plants' : 'Releves_Climatiques'); if (!sheet) return []; var rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return []; var headers = rows[0]; var result = []; for (var i = 1; i < rows.length; i++) { var row = rows[i]; var rowDate = new Date(row[0]); var rowSerre = row[3]; if (serreFilter && serreFilter !== 'ALL' && rowSerre !== serreFilter) continue; if (startDate && new Date(startDate) > rowDate) continue; if (endDate && new Date(endDate + 'T23:59:59') < rowDate) continue; var obj = {}; for (var h = 0; h < headers.length; h++) { obj[headers[h]] = row[h]; } result.push(obj); } return result; } // Fonction utilitaire garantissant que les onglets existent toujours function getOrCreateSheet(ss, sheetName, headers) { var sheet = ss.getSheetByName(sheetName); if (!sheet) { sheet = ss.insertSheet(sheetName); sheet.appendRow(headers); sheet.getRange(1, 1, 1, headers.length).setBackground('#064e3b').setFontColor('#FFFFFF').setFontWeight('bold'); sheet.setFrozenRows(1); } return sheet; }
